-- =====================================================
-- 9. FUNGSI PELANGGAN MENGAMBIL NOMOR
-- =====================================================

CREATE OR REPLACE FUNCTION public.buat_antrean(
    p_nama TEXT,
    p_layanan TEXT
)
RETURNS TEXT
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
    v_nomor INTEGER;
    v_antrean TEXT;
    v_nama TEXT;
BEGIN
    v_nama := TRIM(p_nama);

    IF v_nama IS NULL OR v_nama = '' THEN
        RAISE EXCEPTION 'Nama wajib diisi';
    END IF;

    IF LENGTH(v_nama) > 100 THEN
        RAISE EXCEPTION 'Nama maksimal 100 karakter';
    END IF;

    IF p_layanan NOT IN ('R1','R2','R3','R4','PBB','LGL','PPJB') THEN
        RAISE EXCEPTION 'Layanan tidak valid';
    END IF;

    UPDATE public.counter_antrean
    SET nomor_terakhir = nomor_terakhir + 1
    WHERE kode_layanan = p_layanan
    RETURNING nomor_terakhir INTO v_nomor;

    IF v_nomor IS NULL THEN
        RAISE EXCEPTION 'Counter layanan tidak ditemukan';
    END IF;

    v_antrean := p_layanan || '-' || LPAD(v_nomor::TEXT, 3, '0');

    INSERT INTO public.antrean (
        nomor_antrean,
        nama,
        kode_layanan,
        status
    )
    VALUES (
        v_antrean,
        v_nama,
        p_layanan,
        'Menunggu'
    );

    UPDATE public.display_counter
    SET nomor_terakhir = v_nomor
    WHERE kode_layanan = p_layanan;

    RETURN v_antrean;
END;
$$;


-- =====================================================
-- 10. FUNGSI PANGGIL ANTREAN BERIKUTNYA
-- =====================================================

CREATE OR REPLACE FUNCTION public.panggil_berikutnya()
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
    v_aktif BIGINT;
    v_pilihan BIGINT;
    v_nomor TEXT;
    v_nama TEXT;
    v_layanan TEXT;
BEGIN
    IF auth.uid() IS NULL THEN
        RAISE EXCEPTION 'Akses operator diperlukan';
    END IF;

    SELECT queue_id
    INTO v_aktif
    FROM public.active_queue
    WHERE id = 1
    FOR UPDATE;

    IF v_aktif IS NOT NULL THEN
        UPDATE public.antrean
        SET status = 'Menunggu'
        WHERE id = v_aktif
          AND status = 'Dipanggil';
    END IF;

    SELECT id
    INTO v_pilihan
    FROM public.antrean
    WHERE status = 'Menunggu'
    ORDER BY created_at ASC, id ASC
    LIMIT 1
    FOR UPDATE SKIP LOCKED;

    IF v_pilihan IS NULL THEN
        UPDATE public.active_queue
        SET queue_id = NULL
        WHERE id = 1;

        UPDATE public.display_antrean
        SET nomor_antrean = NULL,
            nama = NULL,
            kode_layanan = NULL,
            updated_at = NOW()
        WHERE id = 1;

        RETURN FALSE;
    END IF;

    UPDATE public.antrean
    SET status = 'Dipanggil'
    WHERE id = v_pilihan
    RETURNING nomor_antrean, nama, kode_layanan
    INTO v_nomor, v_nama, v_layanan;

    UPDATE public.active_queue
    SET queue_id = v_pilihan
    WHERE id = 1;

    UPDATE public.display_antrean
    SET nomor_antrean = v_nomor,
        nama = v_nama,
        kode_layanan = v_layanan,
        updated_at = NOW()
    WHERE id = 1;

    RETURN TRUE;
END;
$$;


-- =====================================================
-- 11. FUNGSI PANGGIL ANTREAN BERDASARKAN NOMOR
-- =====================================================

CREATE OR REPLACE FUNCTION public.panggil_antrean(
    p_nomor TEXT
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
    v_aktif BIGINT;
    v_pilihan BIGINT;
    v_nomor TEXT;
    v_nama TEXT;
    v_layanan TEXT;
BEGIN
    IF auth.uid() IS NULL THEN
        RAISE EXCEPTION 'Akses operator diperlukan';
    END IF;

    SELECT queue_id
    INTO v_aktif
    FROM public.active_queue
    WHERE id = 1
    FOR UPDATE;

    SELECT id
    INTO v_pilihan
    FROM public.antrean
    WHERE nomor_antrean = p_nomor
      AND status IN ('Menunggu', 'Dipanggil', 'Dilewati')
    FOR UPDATE;

    IF v_pilihan IS NULL THEN
        RETURN FALSE;
    END IF;

    IF v_aktif IS NOT NULL AND v_aktif <> v_pilihan THEN
        UPDATE public.antrean
        SET status = 'Menunggu'
        WHERE id = v_aktif
          AND status = 'Dipanggil';
    END IF;

    UPDATE public.antrean
    SET status = 'Dipanggil'
    WHERE id = v_pilihan
    RETURNING nomor_antrean, nama, kode_layanan
    INTO v_nomor, v_nama, v_layanan;

    UPDATE public.active_queue
    SET queue_id = v_pilihan
    WHERE id = 1;

    UPDATE public.display_antrean
    SET nomor_antrean = v_nomor,
        nama = v_nama,
        kode_layanan = v_layanan,
        updated_at = NOW()
    WHERE id = 1;

    RETURN TRUE;
END;
$$;

CREATE OR REPLACE FUNCTION public.reset_antrean()
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
    IF auth.uid() IS NULL THEN
        RAISE EXCEPTION 'Akses operator diperlukan';
    END IF;

    DELETE FROM public.antrean;

    UPDATE public.counter_antrean
    SET nomor_terakhir = 0;

    UPDATE public.display_counter

    WHERE id = 1;

    SET nomor_antrean = NULL,
        nama = NULL,
        kode_layanan = NULL,
